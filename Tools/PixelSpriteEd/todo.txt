FORMAT EXPORT
	- version (1ub)
	- export flags (6ub = 2ub ce exporta + 4ub pe ce dimensiuni)
	- nr imagini (1ub)
		- numele imagine [*]
	- MODULES Count (2ub)  [*]
		- MODULE imgID (1ub)[*]
		- x, y, w, h (*b)
	- FMODULES Count (2ub)[*]
		- moduleIDx (*b)
		- OX, OY (*b)
		- flipFlags	(*b)[*]
	- FRAMES Count   (2ub)[*]
		- FModules Count (1ub)
			- FModule IDx (*ub)
		- BBox X, Y, W, H (*b)[*]
		- HitPts Count (1b)[*]
			- X, Y   (*b)
			- flags	 (*b) [*]
	- AFRAMES Count (2ub)[*]
		- Frame IDx   (*ub)
		- Duration	 (*ub)	[*]
		- MoveX, MoveY (*b) [*]
		- flags		   (*ub)[*]
	- ANIMS Count  (2ub)[*]
		- Anim Flags (*ub)
		- Aframes Count	(1ub)
			- Aframes IDx  (*ub)

EXPLICATII EXPORT
	(2b) = 2 bytes; (*b) = select din form export pe cati bytes			
	(2ub) = 2 bytes fara semn
	[*] = checkbox suprimare in export form
